let currentQuestion = 1;
let responses = {
    firstName: '',
    lastName: '',
    birthdate: '',
    gender: ''
};

const questions = [
    "What's your first name?",
    "What's your last name?",
    "What's your date of birth?",
    "What is your gender?"
];

function handleResponse() {
    const userInput = document.getElementById('userInput').value.trim();
    if (userInput === '') return;

    showUserResponse(userInput);
    saveResponse(currentQuestion, userInput);

    document.getElementById('userInput').value = '';
    if (currentQuestion < questions.length) {
        currentQuestion++;
        setTimeout(() => showBotQuestion(questions[currentQuestion - 1]), 500);
    } else {
        submitForm();
    }
}

function showUserResponse(response) {
    const messageDiv = document.createElement('div');
    messageDiv.className = 'message user-message';
    messageDiv.textContent = response;
    document.querySelector('.chat-container').appendChild(messageDiv);
}

function showBotQuestion(question) {
    const botMessageDiv = document.createElement('div');
    botMessageDiv.className = 'message bot-message';
    botMessageDiv.innerHTML = `<span class="typing-effect">${question}</span>`;
    document.querySelector('.chat-container').appendChild(botMessageDiv);
}

function saveResponse(step, response) {
    switch (step) {
        case 1:
            document.getElementById('firstNameField').value = response;
            responses.firstName = response;
            break;
        case 2:
            document.getElementById('lastNameField').value = response;
            responses.lastName = response;
            break;
        case 3:
            document.getElementById('birthdateField').value = response;
            responses.birthdate = response;
            break;
        case 4:
            document.getElementById('genderField').value = response;
            responses.gender = response;
            break;
    }
}

function submitForm() {
    document.getElementById('inscriptionForm').submit();
}
